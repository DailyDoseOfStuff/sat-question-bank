# SAT Question Bank UI Pass — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Land users on the Dashboard, replace the filter selects with oneprep-style multiselect dropdowns, add a Check button with an optional retry mode, a three-colour highlighter, a settings page, and fix the dark-mode and correct-answer colour problems.

**Architecture:** The app is one file, `public/index.html` (~2,064 lines): markup, a `<style>` block and one IIFE holding all state and rendering. Changes stay in that file except for one new pair of Worker routes and one migration. There is no build step, no bundler and no test runner — verification is driving the real app in the Browser pane and asserting on the rendered DOM, which is how every prior pass on this project was verified.

**Tech Stack:** Cloudflare Worker (`src/index.js`), D1 (SQLite), Supabase auth, vanilla JS + CSS custom properties, KaTeX from CDN, `wrangler dev` for local.

**Spec:** `docs/superpowers/specs/2026-09-03-ui-fixes-design.md`

```js
// ---------------- settings ----------------
// One object, one storage key. These change shape often; a column per toggle is
// churn for nothing. Guests keep them in localStorage, accounts on the server.
const SET_DEFAULTS = { theme: 'system', retry: false, rand: false, count: 0, ai: '' };
let SET = Object.assign({}, SET_DEFAULTS, LS.get('settings', {}));

function resolvedTheme() {
  return SET.theme === 'system'
    ? (matchMedia('(prefers-color-scheme: dark)').matches ? 'dark' : 'light')
    : SET.theme;
}

function applyTheme() {
  const t = resolvedTheme();
  document.documentElement.dataset.theme = t;
  document.querySelectorAll('[data-theme-toggle]').forEach(b => b.textContent = t === 'dark' ? '☀️' : '🌙');
}

function saveSettings() {
  LS.set('settings', SET);
  if (!authToken) return;
  fetch('/api/settings', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json', ...sbHeaders() },
    body: JSON.stringify(SET)
  }).catch(() => {});
}

// A change to one setting: write it, apply it, redraw whatever reads it.
function setSetting(k, v) {
  SET[k] = v;
  saveSettings();
  if (k === 'theme') applyTheme();
  if (k === 'count' || k === 'rand') drawFilters();
}

document.querySelectorAll('[data-theme-toggle]').forEach(b => b.onclick = () =>
  setSetting('theme', resolvedTheme() === 'dark' ? 'light' : 'dark'));

matchMedia('(prefers-color-scheme: dark)').addEventListener('change', () => {
  if (SET.theme === 'system') applyTheme();
});

applyTheme();
```

```js
// ---------------- filter dropdown ----------------
// One control, three screens. `opts` is a flat list of { v, label, group } where a
// group header is its own row that toggles every option under it. `sel` is null
// for "everything" - the same convention F.skills already uses, so a first-time
// visitor is never filtered down to nothing.
function ddSummary(opts, sel, allLabel) {
  if (!sel) return allLabel;
  if (!sel.length) return 'None selected';
  const names = opts.filter(o => !o.head && sel.includes(o.v)).map(o => o.label);
  if (names.length === opts.filter(o => !o.head).length) return allLabel;
  return names.length <= 2 ? names.join(', ') : names.length + ' selected';
}

function dd(host, cfg) {
  // cfg: { label, allLabel, opts:[{v,label,head,group}], get():sel|null, set(sel) }
  const sel = cfg.get();
  const set = sel ? new Set(sel) : null;
  const on = (v) => !set || set.has(v);
  const box = (state) => `<span class="box ${state}">${TICK}</span>`;

  host.className = 'dd' + (sel && sel.length !== cfg.opts.filter(o => !o.head).length ? ' set' : '');
  host.innerHTML =
    `<button class="dd-t" type="button">
       <span><span class="dd-lab">${esc(cfg.label)}</span><span class="dd-val">${esc(ddSummary(cfg.opts, sel, cfg.allLabel))}</span></span>${CHEV}
     </button>`;

  host.querySelector('.dd-t').onclick = (e) => {
    e.stopPropagation();
    const wasOpen = host.classList.contains('open');
    document.querySelectorAll('.dd.open').forEach(d => { d.classList.remove('open'); const p = d.querySelector('.dd-p'); if (p) p.remove(); });
    if (wasOpen) return;
    host.classList.add('open');
    const panel = document.createElement('div');
    panel.className = 'dd-p';
    panel.innerHTML = cfg.opts.map(o => {
      if (o.head) {
        const kids = cfg.opts.filter(k => k.group === o.v && !k.head).map(k => k.v);
        const nOn = kids.filter(on).length;
        return `<button class="dd-o" type="button" data-v="${esc(o.v)}" data-head="1">${box(nOn === kids.length ? 'on' : nOn ? 'some' : '')}<b>${esc(o.label)}</b></button>`;
      }
      return `<button class="dd-o${o.group ? ' sub' : ''}" type="button" data-v="${esc(o.v)}">${box(on(o.v) ? 'on' : '')}<b>${esc(o.label)}</b></button>`;
    }).join('') + `<div class="dd-sep"></div>
      <button class="dd-o" type="button" data-all="1">${box('')}<b>Select all</b></button>
      <button class="dd-o" type="button" data-none="1">${box('')}<b>Clear</b></button>`;
    host.appendChild(panel);
    panel.onclick = (ev) => {
      const b = ev.target.closest('.dd-o');
      if (!b) return;
      ev.stopPropagation();
      const all = cfg.opts.filter(o => !o.head).map(o => o.v);
      if (b.dataset.all) return cfg.set(null);
      if (b.dataset.none) return cfg.set([]);
      const next = new Set(sel || all);
      const touch = b.dataset.head
        ? cfg.opts.filter(k => k.group === b.dataset.v && !k.head).map(k => k.v)
        : [b.dataset.v];
      const allOn = touch.every(v => next.has(v));
      touch.forEach(v => allOn ? next.delete(v) : next.add(v));
      // Everything ticked is the same as nothing ticked, and does not go stale
      // the next time the bank grows.
      cfg.set(next.size === all.length ? null : [...next]);
    };
  };
}

document.addEventListener('click', () => {
  document.querySelectorAll('.dd.open').forEach(d => { d.classList.remove('open'); const p = d.querySelector('.dd-p'); if (p) p.remove(); });
});

});

function grade(q) {
  const ok = isRight(q, S.ans[q.id]);
  const first = !S.tried[q.id];
  S.tried[q.id] = true;

  // Retry mode: a wrong answer marks only the choice that was picked and leaves
  // the question open. Nothing turns green that the student did not choose.
  const keepOpen = SET.retry && ok === false;
  if (keepOpen) {
    S.miss[q.id] = S.miss[q.id] || [];
    if (!S.miss[q.id].includes(S.ans[q.id])) S.miss[q.id].push(S.ans[q.id]);
    S.sel[q.id] = null;
  } else {
    S.checked[q.id] = true;
  }

  if (ok === null) { renderAnswerArea(q); return; }

  const now = new Date().toISOString();
  const ms = Math.round(Date.now() - S.qStart);

  // Only the first Check on a question moves the record. Otherwise a student
  // walks any question to Green by clicking every option in turn.
  if (first) {
    const p = PROG[q.id] || { question_id: q.id, attempts:0, corrects:0 };
    p.attempts++;
    p.corrects += (ok === true ? 1 : 0);
    const wasWrong = p.marker === 'Red';
    p.marker = ok === true ? (wasWrong ? 'Orange' : 'Green') : 'Red';
    p.last_reviewed = now;
    p.time_taken_ms = ms;
    PROG[q.id] = p;
    saveProgress([p]);
  }

  // The attempt log is the history, and a retry is work that was really done.
  const ev = { question_id: q.id, ts: now, correct: ok === true ? 1 : 0, time_taken_ms: ms };
  LOG.push(ev);
  saveLog([ev]);

  renderAnswerArea(q);
}
```
